cmake_minimum_required(VERSION 3.28)
project(fzvs_server VERSION 0.2.0 LANGUAGES C)
add_library(fzvs-server-core STATIC src/server.c)
target_include_directories(fzvs-server-core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include" "${CMAKE_CURRENT_SOURCE_DIR}/../shared")
set_target_properties(fzvs-server-core PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
target_compile_definitions(fzvs-server-core PRIVATE _POSIX_C_SOURCE=200809L)
add_executable(fzvs-server src/main.c)
target_link_libraries(fzvs-server PRIVATE fzvs-server-core)
set_target_properties(fzvs-server PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
option(FZVS_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
foreach(target fzvs-server-core fzvs-server)
  if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wformat=2)
    if(FZVS_SANITIZERS)
      target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${target} PUBLIC -fsanitize=address,undefined)
    endif()
  endif()
endforeach()
